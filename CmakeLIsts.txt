cmake_minimum_required(VERSION 3.10)
project(ChatAppCLI)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

set(INCLUDE_DIR ${CMAKE_SOURCE_DIR}/include)
set(SRC_DIR ${CMAKE_SOURCE_DIR}/src)


# Add include directory for all targets
include_directories(${INCLUDE_DIR})

set(COMMON_SOURCES
    ${SRC_DIR}/User.cpp
    ${SRC_DIR}/Message.cpp
    ${SRC_DIR}/TextMessage.cpp
    ${SRC_DIR}/ChatRoom.cpp
    ${SRC_DIR}/Server.cpp
    ${SRC_DIR}/NetworkManager.cpp
    ${SRC_DIR}/Serializer.cpp
    ${SRC_DIR}/UserManager.cpp
    ${SRC_DIR}/ChatRoomManager.cpp
    ${SRC_DIR}/DMManager.cpp
    ${SRC_DIR}/BlockingIOStrategy.cpp
    ${SRC_DIR}/SelectStrategy.cpp
    # ${SRC_DIR}/PrivateChatRoom.cpp
    # ${SRC_DIR}/GroupChatRoom.cpp
)


# Server executable
add_executable(server
    ${COMMON_SOURCES}
    ${CMAKE_SOURCE_DIR}/server/main.cpp
)

# Client executable
add_executable(client
    ${CMAKE_SOURCE_DIR}/client/main.cpp
    ${SRC_DIR}/Client.cpp
    ${SRC_DIR}/NetworkManager.cpp
    ${SRC_DIR}/Serializer.cpp
    ${INCLUDE_DIR}/PosixNetworkConnection.h
)

target_include_directories(client PRIVATE ${INCLUDE_DIR})

# Test executable for serializer
add_executable(test_serializer
    ${CMAKE_SOURCE_DIR}/tests/test_serializer.cpp
    ${SRC_DIR}/Serializer.cpp
)

target_include_directories(test_serializer PRIVATE ${INCLUDE_DIR})

# Test executable for network protocol
add_executable(test_network_protocol
    ${CMAKE_SOURCE_DIR}/tests/test_network_protocol.cpp
    ${SRC_DIR}/Serializer.cpp
)

target_include_directories(test_network_protocol PRIVATE ${INCLUDE_DIR})

# Test executable for SelectStrategy
add_executable(test_selectstrategy
    ${CMAKE_SOURCE_DIR}/tests/test_selectstrategy.cpp
    ${COMMON_SOURCES}
)

target_include_directories(test_selectstrategy PRIVATE ${INCLUDE_DIR})
target_link_libraries(test_selectstrategy pthread)
