cmake_minimum_required(VERSION 3.10)
project(ChatAppCLI)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

# Qt6 Configuration
find_package(Qt6 REQUIRED COMPONENTS Widgets)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
set(CMAKE_AUTOUIC ON)

set(INCLUDE_DIR ${CMAKE_SOURCE_DIR}/include)
set(SRC_DIR ${CMAKE_SOURCE_DIR}/src)


# Add include directory for all targets
include_directories(${INCLUDE_DIR})

# Network layer sources
set(NETWORK_SOURCES
    ${SRC_DIR}/network/NetworkManager.cpp
    ${SRC_DIR}/network/PosixNetworkConnection.cpp
)

# Protocol layer sources
set(PROTOCOL_SOURCES
    ${SRC_DIR}/protocol/Serializer.cpp
    ${SRC_DIR}/protocol/Message.cpp
    ${SRC_DIR}/protocol/TextMessage.cpp
)

# Domain layer sources
set(DOMAIN_SOURCES
    ${SRC_DIR}/domain/User.cpp
    ${SRC_DIR}/domain/ChatRoom.cpp
)

# Server manager sources
set(SERVER_MANAGER_SOURCES
    ${SRC_DIR}/server/managers/UserManager.cpp
    ${SRC_DIR}/server/managers/ChatRoomManager.cpp
    ${SRC_DIR}/server/managers/DMManager.cpp
)

# Server strategy sources
set(SERVER_STRATEGY_SOURCES
    ${SRC_DIR}/server/strategies/BlockingIOStrategy.cpp
    ${SRC_DIR}/server/strategies/SelectStrategy.cpp
)

# Server application sources
set(SERVER_APP_SOURCES
    ${SRC_DIR}/server/Server.cpp
)

# Client application sources
set(CLIENT_APP_SOURCES
    ${SRC_DIR}/client/Client.cpp
)

# Client strategy sources
set(CLIENT_STRATEGY_SOURCES
    ${SRC_DIR}/client/strategies/BlockingClientStrategy.cpp
)

# Qt-specific strategy sources
set(CLIENT_STRATEGY_SOURCES_QT
    ${SRC_DIR}/client/strategies/EventDrivenClientStrategy.cpp
    ${INCLUDE_DIR}/client/strategies/EventDrivenClientStrategy.h
)

# GUI application sources
set(GUI_SOURCES
    ${SRC_DIR}/gui/MainWindow.cpp
    ${INCLUDE_DIR}/gui/MainWindow.h
    ${SRC_DIR}/gui/ClientController.cpp
    ${INCLUDE_DIR}/gui/ClientController.h
    ${SRC_DIR}/gui/NetworkWorker.cpp
    ${INCLUDE_DIR}/gui/NetworkWorker.h
)

# Common sources for server
set(COMMON_SERVER_SOURCES
    ${NETWORK_SOURCES}
    ${PROTOCOL_SOURCES}
    ${DOMAIN_SOURCES}
    ${SERVER_MANAGER_SOURCES}
    ${SERVER_STRATEGY_SOURCES}
    ${SERVER_APP_SOURCES}
)


# Server executable
add_executable(server
    ${COMMON_SERVER_SOURCES}
    ${CMAKE_SOURCE_DIR}/server/main.cpp
)

# Client executable
add_executable(client
    ${CMAKE_SOURCE_DIR}/client/main.cpp
    ${CLIENT_APP_SOURCES}
    ${CLIENT_STRATEGY_SOURCES}
    ${NETWORK_SOURCES}
    ${PROTOCOL_SOURCES}
)

target_include_directories(client PRIVATE ${INCLUDE_DIR})

# Test executable for serializer
add_executable(test_serializer
    ${CMAKE_SOURCE_DIR}/tests/test_serializer.cpp
    ${PROTOCOL_SOURCES}
)

target_include_directories(test_serializer PRIVATE ${INCLUDE_DIR})

# GUI Client executable
add_executable(client_gui
    ${CMAKE_SOURCE_DIR}/client/main_gui.cpp
    ${CLIENT_STRATEGY_SOURCES}
    ${CLIENT_APP_SOURCES}    ${CLIENT_STRATEGY_SOURCES_QT}    ${GUI_SOURCES}
    ${NETWORK_SOURCES}
    ${PROTOCOL_SOURCES}
)

target_include_directories(client_gui PRIVATE ${INCLUDE_DIR})
target_link_libraries(client_gui Qt6::Widgets)
